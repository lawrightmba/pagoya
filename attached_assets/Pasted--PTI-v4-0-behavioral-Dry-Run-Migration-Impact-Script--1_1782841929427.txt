/**
 * PTI v4.0-behavioral Dry-Run / Migration Impact Script
 * ------------------------------------------------------
 * Purpose: Before flipping v4.0-behavioral live, batch-compute it against
 * every existing user's CURRENT data, diff against their last stored v3.0
 * score, and surface anyone who crosses a meaningful threshold purely
 * because of the model change (not because their behavior changed).
 *
 * ASSUMPTIONS TO VERIFY / ADJUST BEFORE RUNNING:
 *   1. You have a `users` table with an `id` (or `user_id`) primary key.
 *   2. You have a `pti_scores` (or similar) table storing the most recent
 *      computed score + model_version per user. Adjust table/column names
 *      to match your actual Drizzle schema.
 *   3. There exists an internal function (not the HTTP route) that computes
 *      PTI for a given userId + model version, e.g. `computePTI(userId, version)`.
 *      If compute-now is HTTP-only today, you'll want to refactor the route
 *      handler to call a shared internal function so this script can call it
 *      directly without going through HTTP for every user (faster, no auth
 *      overhead, no rate limiting against yourself).
 *   4. Reward tier thresholds and the READY gate match what's in readinessGate.ts:
 *      Bronce 30 / Plata 50 / Oro 70 / Élite 85 / READY >= 80 + 5 gates.
 *      Update TIER_THRESHOLDS below if these have changed.
 *   5. This script is READ + LOG only. It does not write new scores to
 *      production tables. It writes results to a local CSV/JSON for review.
 *
 * Run with: npx tsx pti-v4-dryrun.ts
 * (or ts-node, depending on your Replit setup)
 */

import { db } from "./db"; // ADJUST: path to your existing Drizzle db client
import { users, ptiScores } from "./schema"; // ADJUST: path to your Drizzle schema
import { eq, desc } from "drizzle-orm";
import { computePTI } from "./pti/compute"; // ADJUST: path to your internal compute function
import * as fs from "fs";

// ---- CONFIG ----------------------------------------------------------

const TIER_THRESHOLDS = [
  { name: "Bronce", min: 30 },
  { name: "Plata", min: 50 },
  { name: "Oro", min: 70 },
  { name: "Élite", min: 85 },
];

const READY_GATE = 80;

// How big a swing (in points) counts as "notable" even if no tier was crossed.
// Useful for catching cases where someone's score moved a lot but stayed
// inside the same tier band — still worth a human glance.
const NOTABLE_SWING_THRESHOLD = 5;

// ---- TYPES -------------------------------------------------------------

interface DryRunResult {
  userId: string;
  v3Score: number | null;
  v4Score: number | null;
  delta: number | null;
  v3Tier: string;
  v4Tier: string;
  tierChanged: boolean;
  crossedReadyGate: boolean; // true if v3 < 80 <= v4, or v3 >= 80 > v4
  notableSwing: boolean;
  flags: string[]; // human-readable notes, e.g. "cold-start: no device history"
}

// ---- HELPERS -------------------------------------------------------------

function tierFor(score: number | null): string {
  if (score === null) return "Unscored";
  let tier = "None";
  for (const t of TIER_THRESHOLDS) {
    if (score >= t.min) tier = t.name;
  }
  return tier;
}

function crossedReady(v3: number | null, v4: number | null): boolean {
  if (v3 === null || v4 === null) return false;
  const wasReady = v3 >= READY_GATE;
  const isReady = v4 >= READY_GATE;
  return wasReady !== isReady;
}

// ---- MAIN ----------------------------------------------------------------

async function main() {
  console.log("Starting PTI v3.0 -> v4.0-behavioral dry run...");

  const allUsers = await db.select({ id: users.id }).from(users);
  console.log(`Found ${allUsers.length} users to evaluate.`);

  const results: DryRunResult[] = [];
  let processed = 0;
  let errors = 0;

  for (const u of allUsers) {
    const flags: string[] = [];

    try {
      // Pull most recent stored v3.0 score for this user.
      const lastV3 = await db
        .select()
        .from(ptiScores)
        .where(eq(ptiScores.userId, u.id))
        // ADJUST: filter by model_version = 'v3.0' if both versions can coexist
        // in the same table, e.g. .where(eq(ptiScores.modelVersion, "v3.0"))
        .orderBy(desc(ptiScores.computedAt))
        .limit(1);

      const v3Score = lastV3.length > 0 ? lastV3[0].score : null;

      if (v3Score === null) {
        flags.push("no-prior-v3-score (likely new user, treat as cold-start)");
      }

      // Compute v4.0-behavioral fresh, WITHOUT writing to production tables.
      // If your computePTI function writes to the DB as a side effect,
      // you'll need a dry-run flag added to it, or compute against a
      // shadow/staging table instead.
      const v4Result = await computePTI(u.id, "v4.0-behavioral", {
        dryRun: true, // ADJUST: ensure your compute function supports this,
                       // or stub a read-only variant for this script
      });

      const v4Score = v4Result?.totalScore ?? null;

      // Cold-start checks: flag users where key v4 sub-signals have no
      // underlying data yet, since this is the scenario we're most worried
      // about silently penalizing.
      if (v4Result?.breakdown) {
        const b = v4Result.breakdown;
        if (b.device_consistency_score === 0 && b.device_consistency_days_observed < 7) {
          flags.push("device_consistency: <7 days observed, score may be artificially low");
        }
        if (b.recovery_score === null || b.recovery_score === undefined) {
          flags.push("recovery_score: no 30+ day gap ever recorded (expected null, confirm not defaulting to 0)");
        }
        if (b.routine_score === 0 && b.routine_days_observed < 14) {
          flags.push("routine_score: <14 days observed, variance calc may be unreliable");
        }
        if (b.financial_curiosity_index === 0 && v4Result.totalPaulaMessages < 3) {
          flags.push("financial_curiosity_index: <3 Paula messages total, ratio may be noisy");
        }
      }

      const delta = v3Score !== null && v4Score !== null ? v4Score - v3Score : null;
      const v3Tier = tierFor(v3Score);
      const v4Tier = tierFor(v4Score);
      const tierChanged = v3Tier !== v4Tier;
      const crossedReadyGate = crossedReady(v3Score, v4Score);
      const notableSwing = delta !== null && Math.abs(delta) >= NOTABLE_SWING_THRESHOLD;

      if (crossedReadyGate) {
        flags.push("CRITICAL: crosses READY gate (80) purely from model change");
      }
      if (tierChanged) {
        flags.push(`tier change: ${v3Tier} -> ${v4Tier}`);
      }

      results.push({
        userId: u.id,
        v3Score,
        v4Score,
        delta,
        v3Tier,
        v4Tier,
        tierChanged,
        crossedReadyGate,
        notableSwing,
        flags,
      });
    } catch (err) {
      errors++;
      console.error(`Error processing user ${u.id}:`, err);
      results.push({
        userId: u.id,
        v3Score: null,
        v4Score: null,
        delta: null,
        v3Tier: "Error",
        v4Tier: "Error",
        tierChanged: false,
        crossedReadyGate: false,
        notableSwing: false,
        flags: [`ERROR: ${err instanceof Error ? err.message : String(err)}`],
      });
    }

    processed++;
    if (processed % 100 === 0) {
      console.log(`Processed ${processed}/${allUsers.length}...`);
    }
  }

  // ---- SUMMARY ----------------------------------------------------------

  const tierChanges = results.filter((r) => r.tierChanged);
  const readyGateCrossings = results.filter((r) => r.crossedReadyGate);
  const notableSwings = results.filter((r) => r.notableSwing);
  const coldStartFlags = results.filter((r) =>
    r.flags.some((f) => f.includes("observed") || f.includes("no-prior") || f.includes("no 30+"))
  );

  const avgDelta =
    results.filter((r) => r.delta !== null).reduce((sum, r) => sum + (r.delta ?? 0), 0) /
    Math.max(1, results.filter((r) => r.delta !== null).length);

  console.log("\n========== DRY RUN SUMMARY ==========");
  console.log(`Total users evaluated: ${results.length}`);
  console.log(`Errors during computation: ${errors}`);
  console.log(`Average score delta (v4 - v3): ${avgDelta.toFixed(2)}`);
  console.log(`Users who changed reward tier: ${tierChanges.length}`);
  console.log(`Users who crossed the READY gate (80): ${readyGateCrossings.length} <-- REVIEW THESE MANUALLY`);
  console.log(`Users with notable swing (>=${NOTABLE_SWING_THRESHOLD} pts, same tier): ${notableSwings.length}`);
  console.log(`Users flagged for cold-start / sparse-data concerns: ${coldStartFlags.length}`);
  console.log("======================================\n");

  // ---- OUTPUT FILES -------------------------------------------------------

  // Full results as JSON for programmatic review
  fs.writeFileSync("pti-v4-dryrun-full.json", JSON.stringify(results, null, 2));

  // CSV for quick spreadsheet review, sorted by absolute delta descending
  const sorted = [...results].sort(
    (a, b) => Math.abs(b.delta ?? 0) - Math.abs(a.delta ?? 0)
  );
  const csvHeader = "userId,v3Score,v4Score,delta,v3Tier,v4Tier,tierChanged,crossedReadyGate,notableSwing,flags\n";
  const csvRows = sorted
    .map((r) =>
      [
        r.userId,
        r.v3Score ?? "",
        r.v4Score ?? "",
        r.delta ?? "",
        r.v3Tier,
        r.v4Tier,
        r.tierChanged,
        r.crossedReadyGate,
        r.notableSwing,
        `"${r.flags.join("; ")}"`,
      ].join(",")
    )
    .join("\n");
  fs.writeFileSync("pti-v4-dryrun-summary.csv", csvHeader + csvRows);

  // Priority review list: anyone crossing the READY gate or a tier boundary
  const priorityReview = results.filter((r) => r.crossedReadyGate || r.tierChanged);
  fs.writeFileSync(
    "pti-v4-dryrun-PRIORITY-REVIEW.json",
    JSON.stringify(priorityReview, null, 2)
  );

  console.log("Output written:");
  console.log("  - pti-v4-dryrun-full.json (all users)");
  console.log("  - pti-v4-dryrun-summary.csv (sorted by impact, spreadsheet-friendly)");
  console.log("  - pti-v4-dryrun-PRIORITY-REVIEW.json (tier/gate crossings only — review before republish)");

  if (readyGateCrossings.length > 0) {
    console.log(
      `\n⚠️  ${readyGateCrossings.length} user(s) would cross the READY gate purely from this model change. ` +
      `Recommend manual review and a migration/grace-period plan before going live.`
    );
  }
}

main()
  .then(() => {
    console.log("Dry run complete.");
    process.exit(0);
  })
  .catch((err) => {
    console.error("Dry run failed:", err);
    process.exit(1);
  });
