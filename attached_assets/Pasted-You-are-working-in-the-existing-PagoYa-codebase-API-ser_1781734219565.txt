import { db } from '../db'; // adjust import to match

const KYC_UPGRADE_THRESHOLD_MXN = 3200;

export async function checkAndUpgradeKycTier(
  userId: number | string
): Promise<void> {
  try {
    // Get current user kyc_tier and total completed 
    // transaction value
    const userResult = await db.query(
      `SELECT kyc_tier FROM users WHERE id = $1`,
      [userId]
    );

    if (!userResult.rows.length) return;
    const { kyc_tier } = userResult.rows[0];

    // Only act on simplified users
    if (kyc_tier !== 'simplified') return;

    // Sum all completed payments for this user
    // Adjust table/column names to match existing schema
    const sumResult = await db.query(
      `SELECT COALESCE(SUM(amount_mxn), 0) as total
       FROM bill_payments
       WHERE user_id = $1
       AND status = 'completed'`,
      [userId]
    );

    const total = parseFloat(sumResult.rows[0].total);

    if (total >= KYC_UPGRADE_THRESHOLD_MXN) {
      // Upgrade tier
      await db.query(
        `UPDATE users 
         SET kyc_tier = 'standard', 
             updated_at = NOW()
         WHERE id = $1 
         AND kyc_tier = 'simplified'`,
        [userId]
      );

      // Log to kyc_upgrade_log table (create if needed)
      await db.query(
        `INSERT INTO kyc_upgrade_log 
         (user_id, from_tier, to_tier, trigger_amount_mxn,
          triggered_at)
         VALUES ($1, 'simplified', 'standard', $2, NOW())
         ON CONFLICT DO NOTHING`,
        [userId, total]
      );

      // Send Paula CURP request via existing Twilio 
      // WhatsApp pattern
      await sendPaulaCurpRequest(userId);
    }
  } catch (err) {
    console.error('[KYC] upgrade check failed:', err);
    // Never throw — never block payment flow
  }
}

async function sendPaulaCurpRequest(
  userId: number | string
): Promise<void> {
  try {
    // Get user phone from users table
    const result = await db.query(
      `SELECT phone, first_name FROM users WHERE id = $1`,
      [userId]
    );
    if (!result.rows.length) return;

    const { phone, first_name } = result.rows[0];
    const name = first_name || 'estimado usuario';

    // Use existing Twilio WhatsApp sender pattern
    // Find it with:

    //   ~/workspace/artifacts/api-server/src/ -l
    // Import and use the same function used elsewhere

      `CURP en tu perfil. Es opcional, pero acelera ` +
      `tu progreso hacia el nivel Oro. ¿Te gustaría ` +
      `agregarlo ahora? Responde con tu CURP o escribe ` +
      `"después" si prefieres hacerlo más tarde.`;

    // Use existing WhatsApp send function here

  } catch (err) {
    console.error('[KYC] Paula CURP request failed:', err);
  }
}
