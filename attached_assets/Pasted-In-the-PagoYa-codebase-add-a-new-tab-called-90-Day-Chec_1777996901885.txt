let ckChecks = JSON.parse(localStorage.getItem('ck_checks')||'{}');

function ckSave(){localStorage.setItem('ck_checks',JSON.stringify(ckChecks));}

function ckUpdateStats(){
  const total = CK_DATA.reduce((s,p)=>s+p.items.length,0);
  const done = Object.values(ckChecks).filter(Boolean).length;
  const pct = Math.round(done/total*100);
  document.getElementById('ck-total').textContent=total;
  document.getElementById('ck-done').textContent=done;
  document.getElementById('ck-pct').textContent=pct+'%';
  document.getElementById('ck-fill').style.width=pct+'%';
  const ph1done=CK_DATA[0].items.filter(i=>ckChecks[i.id]).length;
  document.getElementById('ck-ph1r').textContent=CK_DATA[0].items.length-ph1done;
  const critItems=CK_DATA.flatMap(p=>p.items).filter(i=>i.critical);
  document.getElementById('ck-critr').textContent=critItems.filter(i=>!ckChecks[i.id]).length;
}

function ckRender(){
  const list=document.getElementById('ck-list');
  if(!list)return;
  list.innerHTML='';
  CK_DATA.forEach(phase=>{
    const ph=document.createElement('div');
    const done=phase.items.filter(i=>ckChecks[i.id]).length;
    ph.innerHTML=`<div class="ck-phase-header ${phase.phClass}"><div><div class="ck-phase-title">${phase.label}</div><div class="ck-phase-sub">${done}/${phase.items.length} completed</div></div><span class="ck-phase-badge ${phase.badgeClass}">${phase.items.length} actions</span></div>`;
    const itemsDiv=document.createElement('div');
    itemsDiv.className='ck-items';
    phase.items.forEach(item=>{
      const el=document.createElement('div');
      el.className='ck-item';
      const tagHtml=item.tags.map(t=>`<span class="ck-tag ck-tag-${t}">${t}</span>`).join('');
      const critTag=item.critical?'<span class="ck-tag ck-tag-crit">critical</span>':'';
      el.innerHTML=`<div class="ck-item-header"><div class="ck-check${ckChecks[item.id]?' ck-done':''}" data-id="${item.id}"></div><div class="ck-item-main"><div class="ck-item-title">${item.title}</div><div class="ck-item-meta">${tagHtml}${critTag}<span class="ck-week">${item.week}</span></div></div></div><div class="ck-item-body"><div class="ck-body-grid"><div class="ck-body-section"><label>Why this matters</label><p>${item.reasoning}</p></div><div class="ck-body-section"><label>Specific action</label><p>${item.action}</p></div></div><div class="ck-comp"><label>Competitive expected result</label><p>${item.competitive}</p></div></div>`;
      el.querySelector('.ck-check').addEventListener('click',e=>{
        e.stopPropagation();
        ckChecks[item.id]=!ckChecks[item.id];
        ckSave();ckRender();ckUpdateStats();
      });
      el.querySelector('.ck-item-header').addEventListener('click',e=>{
        if(e.target.classList.contains('ck-check'))return;
        el.classList.toggle('ck-open');
      });
      itemsDiv.appendChild(el);
    });
    ph.appendChild(itemsDiv);
    list.appendChild(ph);
  });
}

ckRender();ckUpdateStats();
})();
