  // ── ROADMAP ──
  const RD_KEY = 'cc_roadmap_v1';
  const rdState = JSON.parse(localStorage.getItem(RD_KEY) || '{}');
  const rdCounts = { tech: 8, founder: 8, seo: 7 };

  function rdToggle(item) {
    const cb = item.querySelector('.rd-cb');
    const title = item.querySelector('.rd-title');
    const desc = item.querySelector('.rd-desc');
    const card = item.dataset.card;
    const key = card + '_' + title.textContent.trim().slice(0, 40);
    const done = !cb.classList.contains('done');
    cb.classList.toggle('done', done);
    title.classList.toggle('done', done);
    desc.classList.toggle('done', done);
    rdState[key] = done;
    localStorage.setItem(RD_KEY, JSON.stringify(rdState));
    rdUpdateProgress(card);
  }

  function rdUpdateProgress(card) {
    const items = document.querySelectorAll('.rd-item[data-card="' + card + '"]');
    const done = Array.from(items).filter(i => i.querySelector('.rd-cb.done')).length;
    const total = rdCounts[card];
    const pct = Math.round(done / total * 100);
    document.getElementById('prog-' + card).style.width = pct + '%';
    document.getElementById('prog-' + card + '-label').textContent = done + ' of ' + total + ' complete';
    document.getElementById('prog-' + card + '-pct').textContent = pct + '%';
  }

  function rdInit() {
    document.querySelectorAll('.rd-item').forEach(item => {
      const card = item.dataset.card;
      const key = card + '_' + item.querySelector('.rd-title').textContent.trim().slice(0, 40);
      if (rdState[key]) {
        item.querySelector('.rd-cb').classList.add('done');
        item.querySelector('.rd-title').classList.add('done');
        item.querySelector('.rd-desc').classList.add('done');
      }
    });
    ['tech','founder','seo'].forEach(rdUpdateProgress);
    const dec = new Date('2026-06-01');
    const days = Math.max(0, Math.ceil((dec - new Date()) / 86400000));
    const el = document.getElementById('yc-days-count');
    if (el) el.textContent = days;
  }

  rdInit();
